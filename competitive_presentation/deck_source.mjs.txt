function writeText(slide, text, pos, opts = {}) {
  const shape = slide.shapes.add({
    geometry: "textbox",
    position: pos,
    fill: "none",
    line: { style: "solid", fill: "none", width: 0 },
  });
  shape.text = String(text ?? "");
  shape.text.style = {
    fontSize: opts.size ?? 18,
    bold: opts.bold ?? false,
    color: opts.color ?? C.ink,
    alignment: opts.align ?? "left",
    verticalAlignment: opts.valign ?? "top",
    autoFit: opts.autoFit ?? "shrinkText",
    wrap: "square",
    insets: opts.insets ?? { top: 2, right: 4, bottom: 2, left: 4 },
    typeface: "Aptos",
    lineSpacing: opts.lineSpacing ?? 1.05,
  };
  return shape;
}

function rect(slide, pos, fill, line = "none", radius = false) {
  return slide.shapes.add({
    geometry: radius ? "roundRect" : "rect",
    position: pos,
    fill,
    line: { style: "solid", fill: line, width: line === "none" ? 0 : 1 },
    ...(radius ? { borderRadius: "rounded-xl" } : {}),
  });
}

function line(slide, x1, y1, x2, y2, color = C.line, width = 1) {
  return slide.shapes.add({
    geometry: "line",
    position: { left: x1, top: y1, width: x2 - x1, height: y2 - y1 },
    fill: "none",
    line: { style: "solid", fill: color, width },
  });
}

function chartSegment(slide, x1, y1, x2, y2, color, width = 3) {
  return slide.shapes.add({
    geometry: y2 < y1 ? "lineInv" : "line",
    position: { left: x1, top: Math.min(y1, y2), width: x2 - x1, height: Math.abs(y2 - y1) },
    fill: "none",
    line: { style: "solid", fill: color, width },
  });
}

function baseSlide(title, chapter, subtitle = "", opts = {}) {
  const slide = P.slides.add();
  slide.background.fill = opts.dark ? C.navy : C.pale;
  const page = P.slides.items.length;
  slideMeta.push({ page, title, chapter, kind: opts.kind ?? "main", appendix: opts.appendix ?? null });
  if (opts.targetKey) targetPages[opts.targetKey] = page;
  if (!opts.noHeader) {
    writeText(slide, chapter.toUpperCase(), { left: 64, top: 30, width: 680, height: 24 }, {
      size: 14, bold: true, color: opts.dark ? C.gold : C.teal,
    });
    writeText(slide, title, { left: 64, top: 58, width: 1110, height: 58 }, {
      size: 38, bold: true, color: opts.dark ? C.white : C.navy,
    });
    if (subtitle) {
      writeText(slide, subtitle, { left: 66, top: 114, width: 1100, height: 38 }, {
        size: 18, color: opts.dark ? "#CBD7E4" : C.muted,
      });
    }
    line(slide, 64, 154, 1216, 154, opts.dark ? C.navy2 : C.line, 1);
  }
  writeText(slide, String(page).padStart(2, "0"), { left: 1184, top: 680, width: 40, height: 20 }, {
    size: 12, bold: true, color: opts.dark ? "#91A4B6" : C.muted, align: "right",
  });
  return slide;
}

function addLink(slide, label, pos, targetKey = null, targetPage = null, url = null, back = false, metadata = {}) {
  writeText(slide, label, pos, {
    size: 15, bold: true, color: back ? C.teal : C.navy, align: back ? "left" : "right", valign: "middle",
  });
  linkManifest.push({ page: P.slides.items.length, rect: pos, targetKey, targetPage, url, label, ...metadata });
}

function topicTrendValue(row, metric = "positive_share_pct") {
  return row && row.mentions > 0 && row[metric] != null ? row[metric] : null;
}

function lowScoreNegativeCount(slug, topicName) {
  const row = data.restaurants[slug].low_score.top_negative_topics.find(x => x.topic === topicName);
  return row?.negative_reviews ?? 0;
}

function markerLegend(slide, top = 166) {
  let x = 66;
  for (const slug of SLUGS) {
    rect(slide, { left: x, top: top + 4, width: 12, height: 12 }, COLORS[slug]);

  }
}

function card(slide, pos, title, value, note = "", accent = C.teal, opts = {}) {
  rect(slide, pos, C.white, C.line, true);
  rect(slide, { left: pos.left, top: pos.top, width: 7, height: pos.height }, accent);
  writeText(slide, title, { left: pos.left + 22, top: pos.top + 18, width: pos.width - 40, height: 26 }, {
    size: opts.titleSize ?? 16, bold: true, color: C.muted,
  });
  writeText(slide, value, { left: pos.left + 22, top: pos.top + 49, width: pos.width - 40, height: opts.valueHeight ?? 56 }, {
    size: opts.valueSize ?? 34, bold: true, color: opts.valueColor ?? C.navy,
  });
  if (note) writeText(slide, note, { left: pos.left + 22, top: pos.top + pos.height - 42, width: pos.width - 40, height: 28 }, {
    size: 14, color: C.muted,
  });
}

function noteBox(slide, text, pos, tone = "teal") {
  const fill = tone === "red" ? C.softRed : tone === "gold" ? C.softGold : tone === "blue" ? C.softBlue : C.softTeal;
  const accent = tone === "red" ? C.red : tone === "gold" ? C.gold : tone === "blue" ? C.blue : C.teal;
  rect(slide, pos, fill, "none", true);
  rect(slide, { left: pos.left, top: pos.top, width: 6, height: pos.height }, accent);
  writeText(slide, text, { left: pos.left + 18, top: pos.top + 12, width: pos.width - 32, height: pos.height - 22 }, {
    size: 17, color: C.ink, valign: "middle",
  });
}

function addTable(slide, rows, columns, pos, opts = {}) {
  const headerH = opts.headerH ?? 42;
  const rowH = opts.rowH ?? 42;
  const widths = columns.map(c => c.width);
  let x = pos.left;
  columns.forEach((col, i) => {
    rect(slide, { left: x, top: pos.top, width: widths[i], height: headerH }, C.navy);
    writeText(slide, col.label, { left: x + 6, top: pos.top + 7, width: widths[i] - 12, height: headerH - 12 }, {
      size: opts.headerSize ?? 15, bold: true, color: C.white, align: col.align ?? "left", valign: "middle",
    });
    x += widths[i];
  });
  rows.forEach((row, ri) => {
    const y = pos.top + headerH + ri * rowH;
    x = pos.left;
    columns.forEach((col, ci) => {
      rect(slide, { left: x, top: y, width: widths[ci], height: rowH }, ri % 2 ? "#F8FAFC" : C.white, C.line);
      const value = typeof col.value === "function" ? col.value(row) : row[col.key];
      writeText(slide, value ?? "—", { left: x + 6, top: y + 5, width: widths[ci] - 12, height: rowH - 9 }, {
        size: opts.bodySize ?? 16, bold: col.bold ?? false, color: col.color ? col.color(row) : C.ink,
        align: col.align ?? "left", valign: "middle",
      });
      x += widths[ci];
    });
  });
}

function addBar(slide, categories, series, pos, opts = {}) {
  slide.charts.add("bar", {
    position: pos,
    categories,
    series,
    barOptions: { direction: opts.direction ?? "column", grouping: opts.grouping ?? "clustered", gapWidth: 36 },
    hasLegend: series.length > 1,
    legend: { position: "bottom", overlay: false, textStyle: { fill: C.muted, fontSize: 14 } },
    xAxis: { textStyle: { fill: C.muted, fontSize: 14 }, majorGridlines: null },
    yAxis: {
      min: opts.min, max: opts.max, numberFormatCode: opts.numberFormatCode,
      textStyle: { fill: C.muted, fontSize: 13 },
      majorGridlines: { style: "solid", fill: C.line, width: 1 },
    },
    dataLabels: opts.labels === false ? undefined : {
      showValue: true, position: opts.direction === "bar" ? "outEnd" : "outEnd",
      textStyle: { fill: C.ink, fontSize: 13, bold: true },
    },
  });
}

function addLine(slide, categories, series, pos, opts = {}) {
  slide.charts.add("line", {
    position: pos,
    categories,
    series,
    legend: { position: "bottom", overlay: false, textStyle: { fill: C.muted, fontSize: 14 } },
    yAxis: {
      min: opts.min, max: opts.max, numberFormatCode: opts.numberFormatCode,
      textStyle: { fill: C.muted, fontSize: 13 },
      majorGridlines: { style: "solid", fill: C.line, width: 1 },
    },
    xAxis: { textStyle: { fill: C.muted, fontSize: 13 }, majorGridlines: null },
  });
}

function addGapLineChart(slide, categories, series, pos, opts = {}) {
  const min = opts.min ?? 0;
  const max = opts.max ?? 100;
  const plot = { left: pos.left + 40, top: pos.top + 8, width: pos.width - 50, height: pos.height - 82 };
  for (let tick = min; tick <= max; tick += 20) {
    const y = plot.top + plot.height - ((tick - min) / (max - min)) * plot.height;
    line(slide, plot.left, y, plot.left + plot.width, y, C.line, 1);
    writeText(slide, tick, { left: pos.left, top: y - 10, width: 34, height: 20 }, { size: 12, color: C.muted, align: "right" });
  }
  categories.forEach((category, index) => {
    const x = plot.left + (index / Math.max(1, categories.length - 1)) * plot.width;
    writeText(slide, category, { left: x - 28, top: plot.top + plot.height + 8, width: 56, height: 20 }, { size: 12, color: C.muted, align: "center" });
  });
  series.forEach(item => {
    const color = item.line?.fill ?? C.blue;
    const points = item.values.map((value, index) => {
      if (value == null || !Number.isFinite(Number(value))) return null;
      return {
        x: plot.left + (index / Math.max(1, categories.length - 1)) * plot.width,
        y: plot.top + plot.height - ((Number(value) - min) / (max - min)) * plot.height,
      };
    });
    for (let index = 1; index < points.length; index += 1) {
      if (points[index - 1] && points[index]) {
        chartSegment(slide, points[index - 1].x, points[index - 1].y, points[index].x, points[index].y, color, 3);
      }
    }
    points.filter(Boolean).forEach(point => rect(slide, { left: point.x - 3, top: point.y - 3, width: 6, height: 6 }, color, "none", true));
  });
  series.forEach((item, index) => {
    const cellW = plot.width / series.length;
    const left = plot.left + index * cellW;
    rect(slide, { left, top: pos.top + pos.height - 25, width: 12, height: 4 }, item.line?.fill ?? C.blue);
    writeText(slide, item.name, { left: left + 17, top: pos.top + pos.height - 34, width: cellW - 18, height: 22 }, { size: 12, color: C.muted });
  });
}

function ev(count) { return count >= 15 ? "high" : count >= 5 ? "medium" : count >= 1 ? "low" : "none"; }
function fmt(value, digits = 1) { return value == null ? "—" : Number(value).toFixed(digits); }
function pct(value, digits = 1) { return value == null ? "—" : `${Number(value).toFixed(digits)}%`; }
function euro(value) { return value == null ? "—" : `€${Number(value).toFixed(2)}`; }
function clip(text, n = 100) { const s = String(text ?? "").replace(/\s+/g, " ").trim(); return s.length > n ? `${s.slice(0, n - 1)}…` : s; }
function topic(slug, period, name) { return data.restaurants[slug][period][name] ?? {}; }
function period(slug, key) { return data.restaurants[slug].periods[key]; }

async function writeBlob(filePath, blob) {
  await fs.writeFile(filePath, new Uint8Array(await blob.arrayBuffer()));
}
